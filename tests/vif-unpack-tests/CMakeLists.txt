cmake_minimum_required(VERSION 3.21)
project(vif_unpack_tests LANGUAGES CXX)
set(PS2RECOMP_DIR "" CACHE PATH "Full patched runtime source checkout")
set(HOLYLAND_TESTS_DIR "${CMAKE_CURRENT_LIST_DIR}/.." CACHE PATH "Holyland tests directory")
set(VIF_UNPACK_REFERENCE_PARSER "" CACHE FILEPATH "Optional unoptimized parser for the same authored oracle")
set(RUNTIME "${PS2RECOMP_DIR}/ps2xRuntime")
if(NOT EXISTS "${RUNTIME}/src/lib/ps2_vif1_interpreter.cpp")
    message(FATAL_ERROR "Set PS2RECOMP_DIR to the actual patched candidate source")
endif()
find_package(Python3 3.10 REQUIRED COMPONENTS Interpreter)
include("${HOLYLAND_TESTS_DIR}/cmake/CooperativeRuntimeFixtures.cmake")
set(controls "${CMAKE_CURRENT_BINARY_DIR}/controls")
execute_process(COMMAND "${CMAKE_COMMAND}" -E env PYTHONDONTWRITEBYTECODE=1
    "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/prepare_controls.py"
    --runtime "${RUNTIME}" --output "${controls}" COMMAND_ERROR_IS_FATAL ANY)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
    "${CMAKE_CURRENT_LIST_DIR}/prepare_controls.py" "${RUNTIME}/src/lib/ps2_vif1_interpreter.cpp")
add_library(vif_unpack_memory OBJECT "${RUNTIME}/src/lib/ps2_memory.cpp"
    "${RUNTIME}/src/lib/gs/ps2_gif_arbiter.cpp"
    "${HOLYLAND_TESTS_DIR}/cooperative-dma-tests/memory_stubs.cpp")
cooperative_fixture_flags(vif_unpack_memory)
target_include_directories(vif_unpack_memory PRIVATE "${RUNTIME}/include")
function(unpack_fixture target parser counter)
    add_executable(${target} unpack_fast_tests.cpp "${parser}" $<TARGET_OBJECTS:vif_unpack_memory>)
    cooperative_fixture_flags(${target})
    target_include_directories(${target} PRIVATE "${RUNTIME}/include")
    target_compile_definitions(${target} PRIVATE EXPECT_BULK_UNPACK=${counter})
endfunction()
foreach(kind candidate scalar bad_sign bad_mask bad_skip)
    unpack_fixture(vif_unpack_${kind} "${controls}/${kind}.cpp" 1)
endforeach()
enable_testing()
add_test(NAME vif_unpack_candidate COMMAND vif_unpack_candidate)
add_test(NAME vif_unpack_scalar COMMAND vif_unpack_scalar)
if(VIF_UNPACK_REFERENCE_PARSER)
    unpack_fixture(vif_unpack_reference "${VIF_UNPACK_REFERENCE_PARSER}" 0)
    add_test(NAME vif_unpack_reference COMMAND vif_unpack_reference)
endif()
foreach(kind bad_sign bad_mask bad_skip)
    if(kind STREQUAL "bad_mask")
        set(expected "V4-16 guard admission exact")
    else()
        set(expected "V4-16 destination/full memory exact")
    endif()
    add_test(NAME vif_unpack_${kind}_control COMMAND "${CMAKE_COMMAND}"
        "-DPROGRAM=$<TARGET_FILE:vif_unpack_${kind}>" "-DEXPECTED=${expected}"
        -P "${CMAKE_CURRENT_LIST_DIR}/negative_control.cmake")
endforeach()
get_property(all_tests DIRECTORY PROPERTY TESTS)
set_tests_properties(${all_tests} PROPERTIES TIMEOUT 30)
